    <h2>Cadastro de Produtos</h2>
    <div class="card">
        <input type="text" id="nome" placeholder="Nome do Produto">
        <input type="text" id="descricao" placeholder="Descrição">
        <input type="number" id="preco" placeholder="Preço (Ex: 99.90)" step="0.01">
        <input type="text" id="codigo" placeholder="Código de Barras">
        <button onclick="cadastrarProduto()">Cadastrar Produto</button>
    </div>

    <h3>Lista de Produtos</h3>
    <table id="tabelaProdutos">
        <thead>
            <tr><th>ID</th><th>Nome</th><th>Descrição</th><th>Preço</th><th>Cód. Barras</th></tr>
        </thead>
        <tbody></tbody>
    </table>

    <script>
        const API_URL = 'http://localhost:3000/api/produtos';

        async function carregarProdutos() {
            const res = await fetch(API_URL);
            const dados = await res.json();
            const tbody = document.querySelector('#tabelaProdutos tbody');
            tbody.innerHTML = '';
            dados.forEach(p => {
                tbody.innerHTML += `<tr><td>${p.id}</td><td>${p.nome}</td><td>${p.descricao}</td><td>R$ ${p.preco}</td><td>${p.codigo_barras}</td></tr>`;
            });
        }

        async function cadastrarProduto() {
            const produto = {
                nome: document.getElementById('nome').value,
                descricao: document.getElementById('descricao').value,
                preco: parseFloat(document.getElementById('preco').value),
                codigo_barras: document.getElementById('codigo').value
            };

            await fetch(API_URL, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(produto)
            });

            carregarProdutos();
        }

        carregarProdutos();
    </script>
